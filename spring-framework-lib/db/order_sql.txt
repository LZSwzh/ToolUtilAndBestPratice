CREATE TABLE `tb_order` (
  `order_id` int NOT NULL AUTO_INCREMENT COMMENT '订单ID',
  `order_num` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '订单编号',
  `user_id` int DEFAULT NULL COMMENT '用户ID',
  `order_qty` int DEFAULT NULL COMMENT '订单数量',
  `total_amount` decimal(10,2) DEFAULT NULL COMMENT '总金额',
  `pay_amount` decimal(10,2) DEFAULT NULL COMMENT '实付金额',
  `status` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '付款状态',
  `payment_date` datetime DEFAULT NULL COMMENT '支付时间',
  `ship_date` datetime DEFAULT NULL COMMENT '发货时间',
  `finish_time` datetime DEFAULT NULL COMMENT '完成时间',
  `created_by` int NOT NULL COMMENT '创建人',
  `created_date` datetime NOT NULL COMMENT '单据创建时间',
  `updated_by` int NOT NULL COMMENT '更新人',
  `updated_date` datetime NOT NULL COMMENT '单据更新时间',
  `version` int NOT NULL DEFAULT '0' COMMENT '版本号',
  PRIMARY KEY (`order_id`),
  UNIQUE KEY `uk_order_num` (`order_num`) USING BTREE COMMENT '单据编号唯一索引',
  KEY `idx_user_status` (`user_id`,`status`) COMMENT '用户id和状态联合索引'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='订单头表';

CREATE TABLE `tb_order_item` (
  `order_item_id` int NOT NULL COMMENT '订单明细ID',
  `order_id` int DEFAULT NULL COMMENT '订单头ID',
  `product_id` int DEFAULT NULL COMMENT '产品ID',
  `sku_id` int DEFAULT NULL COMMENT '最小可售单元id',
  `sku_name` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '最小可售单元名称',
  `item_qty` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '订单明细数量',
  `sku_unit_price` decimal(10,2) NOT NULL COMMENT 'sku单价',
  `discount` decimal(10,2) DEFAULT NULL COMMENT '优惠金额',
  `actual_price` decimal(10,2) DEFAULT NULL COMMENT '订单明细实际金额',
  `created_by` int DEFAULT NULL COMMENT '创建人',
  `created_date` datetime DEFAULT NULL COMMENT '创建时间',
  `updated_by` int DEFAULT NULL COMMENT '创建人',
  `update_date` datetime DEFAULT NULL COMMENT '创建时间',
  `version` int NOT NULL DEFAULT '0' COMMENT '版本号',
  PRIMARY KEY (`order_item_id`),
  KEY `idx_order_id` (`order_id`) USING BTREE COMMENT '订单ID索引'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci  COMMENT='订单明细表';

CREATE TABLE `tb_payment_flow` (
  `payment_flow_id` int NOT NULL COMMENT '付款流水ID',
  `payment_flow_num` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '付款流水编码',
  `order_id` int NOT NULL COMMENT '订单ID',
  `third_flow_num` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '支付平台编码',
  `thrd_payment_channel` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '三方支付方式【WECHAT、ALIPAY、BANK】',
  `actual_amount` decimal(10,2) DEFAULT NULL COMMENT '实际支付金额',
  `payment_type` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '支付类型【退款RETURN、付款PAYMENT】',
  `created_by` int NOT NULL COMMENT '创建人',
  `created_date` datetime NOT NULL COMMENT '创建时间',
  `updated_by` int NOT NULL COMMENT '更新人',
  `updated_date` datetime NOT NULL COMMENT '更新时间',
  `version` int NOT NULL DEFAULT '0' COMMENT '版本号',
  PRIMARY KEY (`payment_flow_id`),
  KEY `idx_order_id` (`order_id`) USING BTREE COMMENT '订单索引'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='支付流水表';


CREATE TABLE `tb_order_status_log` (
  `status_log_id` int NOT NULL COMMENT '订单日志ID',
  `order_id` int NOT NULL COMMENT '订单ID',
  `from_status` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '初始状态',
  `to_status` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '更新状态',
  `trigger_type` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '触发类型【USER,SYS,MANUAL,THIRD_SYS,ADMIN】',
  `acted_by` varchar(255) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '操作者【人名、自动、手动、三方系统名、管理员名】',
  `remark` text COLLATE utf8mb4_unicode_ci COMMENT '审计说明',
  `created_date` datetime NOT NULL COMMENT '创建时间',
  `updated_date` datetime NOT NULL COMMENT '修改时间',
  PRIMARY KEY (`status_log_id`),
  KEY `idx_order_id` (`order_id`) USING BTREE COMMENT '订单索引',
  KEY `idx_order_created` (`order_id`,`created_date`) USING BTREE COMMENT '订单和创建时间联合索引'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='订单状态日志';
